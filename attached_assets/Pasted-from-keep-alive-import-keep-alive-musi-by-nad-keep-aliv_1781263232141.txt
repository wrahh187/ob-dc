import asyncio
import random
import os
import discord
from discord.ext import commands
from discord import app_commands
from keep_alive import keep_alive
from unidecode import unidecode

# Intencje
intents = discord.Intents.default()
intents.message_content = True
intents.members = True

bot = commands.Bot(command_prefix='!', intents=intents)

GUILD_ID = 1367084705937358859
ROLE_ID_ALLOWED = 1514940178345496596

# Lista statusów do losowania
statusy = [
    ("Need For Renta 2025", discord.ActivityType.playing),
    ("i je picce 🍕", discord.ActivityType.listening),
    ("i pali papierosa", discord.ActivityType.listening),
    ("i koloruje kolorowanki", discord.ActivityType.watching),
    ("i pije piwo", discord.ActivityType.watching)
]

# Funkcja do zmiany statusu co godzinę
async def zmien_status_co_godzine():
    await bot.wait_until_ready()
    statusy_dostepnosci = [discord.Status.online, discord.Status.idle, discord.Status.dnd]

    while not bot.is_closed():
        tekst, typ = random.choice(statusy)
        dostepnosc = random.choice(statusy_dostepnosci)
        await bot.change_presence(
            status=dostepnosc,
            activity=discord.Activity(type=typ, name=tekst)
        )
        await asyncio.sleep(3600)  # zmienia status co godzinę

# Slash komenda: /say
@bot.tree.command(name="say", description="Bot powie podany tekst", guild=discord.Object(id=GUILD_ID))
@app_commands.describe(message="Co bot ma powiedzieć")
async def say(interaction: discord.Interaction, message: str):
    if not any(role.id == ROLE_ID_ALLOWED for role in interaction.user.roles):
        await interaction.response.send_message(
            "Nie masz uprawnień do użycia tej komendy.",
            ephemeral=True
        )
        return

    await interaction.response.send_message(
        f"✅ Wiadomość została wysłana: `{message}`",
        ephemeral=True
    )

    await interaction.channel.send(message)

# Po zalogowaniu bota
@bot.event
async def on_ready():
    print(f'Zalogowano jako {bot.user}')

    try:
        # Synchronizacja komend globalnych
        global_synced = await bot.tree.sync()
        print(f"Zsynchornizowano {len(global_synced)} globalnych komend slash.")

        # Synchronizacja komend dla konkretnego guilda (np. /say)
        guild_synced = await bot.tree.sync(guild=discord.Object(id=GUILD_ID))
        print(f"Zsynchornizowano {len(guild_synced)} komend slash dla gildii.")
    except Exception as e:
        print(f"Błąd synchronizacji komend: {e}")

    # Uruchomienie pętli zmiany statusu
    bot.loop.create_task(zmien_status_co_godzine())
